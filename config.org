* Fonts
#+begin_src emacs-lisp
  (defun my/setup-fonts ()
    "Setup fonts for coding, reading, and Chinese on different systems"
    (interactive)
    (cond
     ;; MacOS
     ((eq system-type 'darwin)
      (set-face-attribute 'default nil
                          :family "SF Mono"
                          :height 140)
      (set-face-attribute 'fixed-pitch nil
                          :family "SF Mono"
                          :height 140)
      (set-face-attribute 'variable-pitch nil
                          :family "Helvetica"
                          :height 140)

      (set-fontset-font t 'han (font-spec :family "PingFang SC"))
      (set-fontset-font t 'cjk-misc (font-spec :family "PingFang SC")))
     ;; Linux
     ((eq system-type 'gnu/linux)
      (set-face-attribute 'default nil
                          :family "Maple Mono Normal NL NF CN"
                          :height 120)
      (set-face-attribute 'fixed-pitch nil
                          :family "Maple Mono Normal NL NF CN"
                          :height 120)
      (set-face-attribute 'variable-pitch nil
                          :family "Noto Sans"
                          :height 120)

      (set-fontset-font t 'han (font-spec :family "Noto Sans CJK SC"))
      (set-fontset-font t 'cjk-misc (font-spec :family "Noto Sans CJK SC"))
      )
     ;; fallback
     (t
      (message "No font present for system-type: %s" system-type))
     )
    )

  (add-hook 'after-init-hook #'my/setup-fonts)
#+end_src

* Auto tangle config.org
#+begin_src emacs-lisp
  (defun my/maybe-enable-auto-tangle ()
    "Enable auto-tangle only for config.org."
    (let ((bf (buffer-file-name)))
      (when (and bf
                 (string-equal (file-truename bf)
                               (file-truename (expand-file-name "config.org" user-emacs-directory))))
        (add-hook 'after-save-hook #'my/org-auto-tangle-config nil t))))

  (add-hook 'org-mode-hook #'my/maybe-enable-auto-tangle)

  (defun my/reload-config ()
    "Tangle and reload my Emacs config."
    (interactive)
    (let ((org-confirm-babel-evaluate nil)
          (config-org (expand-file-name "config.org" user-emacs-directory))
          (config-el  (expand-file-name "config.el"  user-emacs-directory)))
      (org-babel-tangle-file config-org config-el "emacs-lisp")
      (load-file config-el)
      (message "Reloaded config from %s" config-org)))
#+end_src
